import requests
import json

def emotion_detector(text_to_analyze):
    if text_to_analyze == "":
        return {
            'anger': None,
            'disgust': None,
            'fear': None,
            'joy': None,
            'sadness': None,
            'dominant_emotion': None
        }

    url = 'https://sn-watson-emotion.labs.skills.network/v1/watson.runtime.nlp.v1/NlpService/EmotionPredict'
    myobj = { "raw_document": { "text": text_to_analyze } } 
    header = {"grpc-metadata-mm-model-id": "emotion_aggregated-workflow_lang_en_stock"}
    response = requests.post(url, json = myobj, headers=header)
    formatted_response = json.loads(response.text) 
    
    # Navigates: formatted_response -> 'emotionPredictions' list -> first item -> 'emotion' dictionary
    emotion_scores = formatted_response['emotionPredictions'][0]['emotion']
    
    # Extract the required set of emotions
    anger_score = emotion_scores.get('anger', 0)
    disgust_score = emotion_scores.get('disgust', 0)
    fear_score = emotion_scores.get('fear', 0)
    joy_score = emotion_scores.get('joy', 0)
    sadness_score = emotion_scores.get('sadness', 0)

    result = {
        'anger': anger_score,
        'disgust': disgust_score,
        'fear': fear_score,
        'joy': joy_score,
        'sadness': sadness_score
    }
    
    # Logic to find the dominant emotion (highest score)
    dominant_emotion = max(result, key=result.get)
    
    # Append the dominant emotion to the final output
    result['dominant_emotion'] = dominant_emotion

    return result
